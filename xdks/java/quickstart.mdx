Make your first authenticated X API v2 call with `com.x:xdk` in minutes.

<Steps>
  <Step title="Add com.x:xdk to your project">
```xml
<dependency>
  <groupId>com.x</groupId>
  <artifactId>xdk</artifactId>
  <version>0.1.2</version>
</dependency>
```
  </Step>
  <Step title="Export your Bearer Token">
    Generate a **Bearer Token** in the [X Developer Console](https://console.x.com) and set it in your environment:

```bash
export X_BEARER_TOKEN="your_bearer_token_here"
```
  </Step>
  <Step title="Run your first lookup and search">
```java
import com.x.xdk.Schemas;
import com.x.xdk.XClient;
import com.x.xdk.clients.PostsClient;
import java.util.List;

public class Quickstart {
    public static void main(String[] args) {
        XClient client = XClient.builder()
            .bearerToken(System.getenv("X_BEARER_TOKEN"))
            .build();

        Schemas.GetUsersByUsernameResponse user =
            client.users().getByUsername("XDevelopers");
        System.out.println("User: " + user.getData());

        Schemas.SearchPostsRecentResponse posts = client.posts().searchRecent(
            "from:XDevelopers -is:retweet",
            PostsClient.SearchRecentOptions.builder()
                .maxResults(10)
                .postFields(List.of("created_at", "public_metrics"))
        );
        System.out.println("Posts: " + posts.getData());
    }
}
```
  </Step>
</Steps>
