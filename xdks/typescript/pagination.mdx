Client methods return a single page response. Wrap any paginated call with `UserPaginator`, `PostPaginator`, `EventPaginator`, or the base `Paginator` to follow `next_token` automatically.

## Setup

<CodeGroup dropdown>

```typescript title="paginator.ts" lines wrap icon="square-js" theme={null}
import { Client, UserPaginator, PaginatedResponse, Schemas } from "@xdevplatform/xdk";

const client = new Client({ bearerToken: process.env.X_BEARER_TOKEN });

const followers = new UserPaginator(
  async (token?: string): Promise<PaginatedResponse<Schemas.User>> => {
    const res = await client.users.getFollowers("2244994945", {
      maxResults: 100,
      paginationToken: token,
      userFields: ["id", "name", "username"],
    });
    return {
      data: res.data ?? [],
      meta: res.meta,
      includes: res.includes,
      errors: res.errors,
    };
  }
);
```

```javascript title="paginator.js" lines wrap icon="square-js" theme={null}
import { Client, UserPaginator } from "@xdevplatform/xdk";

const client = new Client({ bearerToken: process.env.X_BEARER_TOKEN });

const followers = new UserPaginator(async (token) => {
  const res = await client.users.getFollowers("2244994945", {
    maxResults: 100,
    paginationToken: token,
    userFields: ["id", "name", "username"],
  });
  return { data: res.data ?? [], meta: res.meta, includes: res.includes, errors: res.errors };
});
```

</CodeGroup>

## Async iteration

<CodeGroup>

```typescript async.ts theme={null}
for await (const user of followers) {
  console.log(user.username);
}
```

```javascript async.js theme={null}
for await (const user of followers) {
  console.log(user.username);
}
```

</CodeGroup>

## Manual paging

<CodeGroup>

```typescript manual.ts theme={null}
await followers.fetchNext(); // first page
while (!followers.done) {
  await followers.fetchNext(); // subsequent pages
}

console.log(followers.users.length);
```

```javascript manual.js theme={null}
await followers.fetchNext();
while (!followers.done) {
  await followers.fetchNext();
}
console.log(followers.items.length);
```

</CodeGroup>

## Error handling

```typescript title="errors.ts" lines wrap icon="square-js" theme={null}
try {
  for await (const user of followers) {
    console.log(user.username);
  }
} catch (err: unknown) {
  if (followers.rateLimited) {
    console.error("Rate limited, retry after backoff");
  } else {
    throw err;
  }
}
```
