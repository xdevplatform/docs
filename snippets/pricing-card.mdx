{/* Pricing tier on an XDS surface. The popular tier is emphasized
    with .x-surface-emphasis (border-active + bg-secondary) and an
    inverted chip; checkmarks use the semantic success color. */}
export const PricingCard = ({
  title,
  price,
  description,
  features = [],
  icon,
  iconType = "solid",
  popular = false,
  href = "https://developer.x.com/en/portal/products"
}) => {
  const isLink = Boolean(href);
  const CardWrapper = isLink ? 'a' : 'div';

  return (
    <CardWrapper
      href={isLink ? href : undefined}
      target={isLink ? "_blank" : undefined}
      rel={isLink ? "noopener noreferrer" : undefined}
      className={`block not-prose ${isLink ? 'cursor-pointer' : ''}`}
    >
      <div className={`x-surface relative p-4 h-full flex flex-col ${popular ? 'x-surface-emphasis' : ''}`}>
        {popular && (
          <div className="absolute top-3 right-3">
            <span className="x-chip-inverted">Popular</span>
          </div>
        )}

        <div className="text-left mb-6">
          <h3 className="x-headline1 x-text-primary mb-2">{title}</h3>

          <div className="mb-3">
            <span className="x-title3 x-text-primary">{price}</span>
            {price !== 'Contact us' && (
              <span className="x-subtext1 x-text-secondary ml-1">/month</span>
            )}
          </div>

          <p className="x-subtext2 x-text-secondary">{description}</p>
        </div>

        {features.length > 0 && (
          <>
            <div className="border-t mb-4" style={{ borderColor: 'var(--x-border-normal)' }}></div>
            <div className="text-left flex-1">
              <h4 className="x-subtext1 x-text-primary mb-3" style={{ fontWeight: 700 }}>
                What's included:
              </h4>
              <div className="space-y-2.5">
                {features.map((feature, index) => (
                  <div key={index} className="flex items-start space-x-2">
                    <svg width="16" height="16" viewBox="0 0 16 16" fill="none" xmlns="http://www.w3.org/2000/svg" className="x-text-success flex-shrink-0 mt-0.5">
                      <path d="M3.11182 8.44444L6.44515 11.7778L14.2229 4" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round"/>
                    </svg>
                    <div className="x-subtext1 x-text-secondary leading-relaxed">{feature}</div>
                  </div>
                ))}
              </div>
            </div>
          </>
        )}
      </div>
    </CardWrapper>
  );
};
